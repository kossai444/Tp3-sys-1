#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>
#include <semaphore.h>

#define BUFFER_SIZE 10
#define COUNT_TO_PRODUCE 20
#define PRODUCERS 3
#define CONSUMERS 3
#define STOP -1

int buffer[BUFFER_SIZE];

int in = 0;
int out = 0;
int next_item = 1;

pthread_mutex_t mutex;

sem_t empty;
sem_t full;


// PRODUCER FUNCTION
void* producer_worker(void* arg)
{
    while (1)
    {
        sem_wait(&empty);

        pthread_mutex_lock(&mutex);

        if (next_item > COUNT_TO_PRODUCE)
        {
            pthread_mutex_unlock(&mutex);
            sem_post(&empty);
            break;
        }

        int item = next_item++;

        buffer[in] = item;

        printf("Producer produced: %d\n", item);

        in = (in + 1) % BUFFER_SIZE;

        pthread_mutex_unlock(&mutex);

        sem_post(&full);
    }

    return NULL;
}


// CONSUMER FUNCTION
void* consumer_worker(void* arg)
{
    while (1)
    {
        sem_wait(&full);

        pthread_mutex_lock(&mutex);

        int item = buffer[out];

        out = (out + 1) % BUFFER_SIZE;

        if (item != STOP)
        {
            printf("Consumer consumed: %d\n", item);
        }

        pthread_mutex_unlock(&mutex);

        sem_post(&empty);

        if (item == STOP)
        {
            break;
        }
    }

    return NULL;
}


// MAIN FUNCTION
int main()
{
    pthread_t producers[PRODUCERS];
    pthread_t consumers[CONSUMERS];

    pthread_mutex_init(&mutex, NULL);

    sem_init(&empty, 0, BUFFER_SIZE);
    sem_init(&full, 0, 0);


    // Create consumers
    for (int i = 0; i < CONSUMERS; i++)
    {
        pthread_create(&consumers[i], NULL,
                       consumer_worker, NULL);
    }


    // Create producers
    for (int i = 0; i < PRODUCERS; i++)
    {
        pthread_create(&producers[i], NULL,
                       producer_worker, NULL);
    }


    // Wait for producers
    for (int i = 0; i < PRODUCERS; i++)
    {
        pthread_join(producers[i], NULL);
    }


    // Send stop signals
    for (int i = 0; i < CONSUMERS; i++)
    {
        sem_wait(&empty);

        pthread_mutex_lock(&mutex);

        buffer[in] = STOP;

        in = (in + 1) % BUFFER_SIZE;

        pthread_mutex_unlock(&mutex);

        sem_post(&full);
    }


    // Wait for consumers
    for (int i = 0; i < CONSUMERS; i++)
    {
        pthread_join(consumers[i], NULL);
    }


    pthread_mutex_destroy(&mutex);

    sem_destroy(&empty);
    sem_destroy(&full);

    printf("\nProgram finished successfully!\n");

    return 0;
}
